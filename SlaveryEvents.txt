country_event = {
    id = 184000
    title = "NEWSLAVES"
    desc = "The ships from Africa have come in, and we've received more slaves for our nation!"
    picture = "slaves_traded"

    trigger = {
        OR = {
            has_country_flag = slave_trade_leader
            has_country_flag = slave_trader
        }
        money = 21000
        NOT = { has_country_modifier = slave_trade }
        OR = { 
                any_country = {
                               limit = { 
                                         OR = { has_country_flag = slave_trader
                                                has_country_flag = slave_trade_leader }  
                                         any_province = {
                                                limit = { continent = central_africa
                                                          continent = west_africa
                                                          continent = south_africa
                                                          continent = east_africa
                                                                                 }  
                                                OR = {  
                                                        owned_by = THIS 
                                                        owned_by = { 
                                                                      any_country = {
                                                                      limit = { 
                                                                                substate_of = THIS
                                                                                vassal_of = THIS
                                                                                }
                                                                      }
                                                                   }
                                                        }
                                                     }
                                                     exists = yes
                                                }
                                        }
                any_country = {
                               limit = { 
                                         capital_scope = { continent = central_africa }
                                         capital_scope = { continent = west_africa }
                                         capital_scope = { continent = south_africa }
                                         capital_scope = { continent = east_africa }
                                         civilized = no
                                         slavery = yes_slavery
                                        }
                               exists = yes
                              }
  
              }
    }

    mean_time_to_happen = {
        days = 7
    }

    option = {
        name = "NEWSLAVESOPTION"
        any_pop = {
            limit = {
                type = slaves
            }
            reduce_pop = 1.008
        }
        money = -2000
        add_country_modifier = {
            name = slave_trade
            duration = 365
        }
        ai_chance = {
            factor = 70
        }
    }

    option = {
        name = "We have more than enough slaves at the moment."
        add_country_modifier = {
            name = slave_trade
            duration = 365
        }
        ai_chance = {
            factor = 30
            modifier = {
                factor = 100
                NOT = { money = 20000 }
            }
        }
    }
}
